cmake_minimum_required(VERSION 3.20)
project(TazeDecompiler VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/luau/Common/include/Luau/Bytecode.h")
    set(TAZE_DEFAULT_LUAU "${CMAKE_CURRENT_SOURCE_DIR}/luau")
else()
    set(TAZE_DEFAULT_LUAU "${CMAKE_CURRENT_SOURCE_DIR}/luau-master/luau-master")
endif()
set(TAZE_LUAU_DIR "${TAZE_DEFAULT_LUAU}" CACHE PATH "Local, untracked Luau source checkout")
if(NOT EXISTS "${TAZE_LUAU_DIR}/Common/include/Luau/Bytecode.h")
    message(FATAL_ERROR "Set TAZE_LUAU_DIR to a Luau checkout. See README.md.")
endif()

add_library(TazeDecompiler src/Bytecode.cpp src/Decompiler.cpp)
target_include_directories(TazeDecompiler PUBLIC include PRIVATE "${TAZE_LUAU_DIR}/Common/include")
if(MSVC)
    target_compile_options(TazeDecompiler PRIVATE /W4 /EHsc)
else()
    target_compile_options(TazeDecompiler PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(taze src/Main.cpp)
target_link_libraries(taze PRIVATE TazeDecompiler)

option(TAZE_BUILD_TESTS "Build compiler/VM round-trip tests and fixture compiler" ON)
if(TAZE_BUILD_TESTS)
    set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
    set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${TAZE_LUAU_DIR}" "luau" EXCLUDE_FROM_ALL)
    enable_testing()
    add_executable(taze-tests tests/Tests.cpp)
    target_link_libraries(taze-tests PRIVATE TazeDecompiler Luau.Compiler Luau.VM)
    target_include_directories(taze-tests PRIVATE "${TAZE_LUAU_DIR}/Common/include")
    add_test(NAME RoundTrip COMMAND taze-tests)
    add_executable(taze-compile tools/Compile.cpp)
    target_link_libraries(taze-compile PRIVATE Luau.Compiler Luau.VM)
endif()
